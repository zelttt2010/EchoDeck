cmake_minimum_required(VERSION 3.25)
project(EchoDeck LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(EchoDeck WIN32 src/main.cpp src/audio_engine.cpp src/audio_engine.h src/bridge_client.cpp src/bridge_client.h src/device_utils.cpp src/device_utils.h src/wav.cpp src/wav.h shared/echodeck_ring.h)
target_compile_definitions(EchoDeck PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(EchoDeck PRIVATE ole32 uuid avrt mmdevapi shell32)
add_executable(EchoDeckCaptureProbe tests/capture_probe.cpp)
target_compile_definitions(EchoDeckCaptureProbe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(EchoDeckBridgeTest tests/bridge_test.cpp src/bridge_client.cpp driver/echodeck_ring_consumer.c)
target_include_directories(EchoDeckBridgeTest PRIVATE src shared driver)
target_compile_definitions(EchoDeckBridgeTest PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(EchoDeckCaptureProbe PRIVATE ole32 uuid)
enable_testing()
add_test(NAME EchoDeckBridgeTest COMMAND EchoDeckBridgeTest)
if(MSVC)
  target_compile_options(EchoDeck PRIVATE /W4 /EHsc)
  target_compile_options(EchoDeckCaptureProbe PRIVATE /W4 /EHsc)
endif()
